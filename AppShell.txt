import { useAuth } from "../features/auth/AuthContext";
import { NavLink, Outlet } from "react-router-dom";

const publicNavItems = [
  { to: "/", label: "Home" },
  { to: "/matches", label: "Matches" },
  { to: "/status", label: "Status" }
];

export function AppShell() {
  const { status, user, signOut } = useAuth();
  const navItems = status === "authenticated"
    ? [
        ...publicNavItems,
        { to: "/dashboard", label: "Dashboard" },
        { to: "/profile", label: "Profile" },
        { to: "/my-predictions", label: "My Predictions" },
        { to: "/leaderboard/tournament", label: "Tournament" }
      ]
    : [...publicNavItems, { to: "/login", label: "Login" }];

  return (
    <div className="shell">
      <header className="topbar">
        <div className="topbar__inner">
          <div className="brand">
            <span className="brand__title">IPL Predictor</span>
            <span className="brand__subtitle">Sprint 10 adds polished prediction UX, internal result entry, automatic scoring orchestration, and cleaner leaderboard experiences on top of the IPL 2026 import and existing gameplay flow.</span>
          </div>
          <div className="topbar__actions">
            <nav className="nav" aria-label="Primary navigation">
              {navItems.map((item) => (
                <NavLink
                  key={item.to}
                  to={item.to}
                  className={({ isActive }) => (isActive ? "nav__link nav__link--active" : "nav__link")}
                >
                  {item.label}
                </NavLink>
              ))}
            </nav>
            {status === "authenticated" ? (
              <div className="session-box">
                <span className="session-box__text">
                  {user?.mobileNumber}
                  {user?.profileCompleted ? " · Profile complete" : " · Profile pending"}
                </span>
                <button className="button button--ghost" type="button" onClick={signOut}>
                  Logout
                </button>
              </div>
            ) : null}
          </div>
        </div>
      </header>
      <main className="page">
        <Outlet />
      </main>
    </div>
  );
}
